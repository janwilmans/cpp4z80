;; Minimal Z80 machine description.

(include "constraints.md")


;; -------------------------------------------------------------------------
;; NOP
;; -------------------------------------------------------------------------

(define_insn "nop"
  [(const_int 0)]
  ""
  "nop")


;; -------------------------------------------------------------------------
;; QI move
;; -------------------------------------------------------------------------

(define_insn "movqi"
  [(set (match_operand:QI 0 "nonimmediate_operand" "=r,r,m")
        (match_operand:QI 1 "general_operand" "r,n,r"))]
  ""
  "@
   ld %0,%1
   ld %0,%1
   ld %0,%1")


;; -------------------------------------------------------------------------
;; HI move
;; -------------------------------------------------------------------------

(define_insn "movhi"
  [(set (match_operand:HI 0 "register_operand" "=r,r")
        (match_operand:HI 1 "general_operand" "r,n"))]
  ""
  "@
   ld %0,%1
   ld %0,%1")


;; -------------------------------------------------------------------------
;; HI addition
;; -------------------------------------------------------------------------

(define_insn "addhi3"
  [(set (match_operand:HI 0 "register_operand" "=r")
        (plus:HI
          (match_operand:HI 1 "register_operand" "0")
          (match_operand:HI 2 "register_operand" "r")))]
  ""
  "add %0,%2")


;; -------------------------------------------------------------------------
;; HI subtraction
;; -------------------------------------------------------------------------

(define_insn "subhi3"
  [(set (match_operand:HI 0 "register_operand" "=r")
        (minus:HI
          (match_operand:HI 1 "register_operand" "0")
          (match_operand:HI 2 "register_operand" "r")))]
  ""
  "sub %0,%2")


;; -------------------------------------------------------------------------
;; Unconditional jump
;; -------------------------------------------------------------------------

(define_insn "jump"
  [(set (pc)
        (label_ref (match_operand 0 "" "")))]
  ""
  "jp %l0")


;; -------------------------------------------------------------------------
;; Return
;; -------------------------------------------------------------------------

(define_insn "return"
  [(return)]
  ""
  "ret")


(define_insn "simple_return"
  [(simple_return)]
  ""
  "ret")
