;; Design notes
;; ------------
;; * The flags register is not modelled.  No instruction pattern relies on
;;   flags set by a previous instruction: compare-and-branch is emitted as a
;;   single insn.  This keeps the port simple and safe under LRA.
;; * Move patterns never clobber registers; when a temporary is needed they
;;   use push/pop or "ex de,hl".  Arithmetic patterns may clobber A.
;; * IX is the frame pointer, IY is a scratch register used only inside
;;   single insns (prologue, stack adjustment, far (ix+d) accesses).

(define_constants
  [(C_REG    0)
   (B_REG    1)
   (E_REG    2)
   (D_REG    3)
   (L_REG    4)
   (H_REG    5)
   (A_REG    6)
   (IXL_REG  7)
   (IYL_REG  9)
   (SP_REG  11)])

(define_c_enum "unspec"
  [UNSPEC_PUSHQI])

(define_c_enum "unspecv"
  [UNSPECV_PROLOGUE
   UNSPECV_EPILOGUE])

(include "constraints.md")

;; =========================================================================
;; NOP
;; =========================================================================

(define_insn "nop"
  [(const_int 0)]
  ""
  "nop")

;; =========================================================================
;; 8-bit moves
;; =========================================================================

(define_expand "movqi"
  [(set (match_operand:QI 0 "nonimmediate_operand")
	(match_operand:QI 1 "general_operand"))]
  ""
{
  if (can_create_pseudo_p ()
      && MEM_P (operands[0])
      && !register_operand (operands[1], QImode))
    operands[1] = force_reg (QImode, operands[1]);
})

;;  r <- r      ld r,r'
;;  r <- n      ld r,n
;;  r <- Q      ld r,(hl) / ld r,(ix+d)
;;  Q <- r      ld (hl),r / ld (ix+d),r
;;  Q <- n      ld (hl),n / ld (ix+d),n
;;  a <- S      ld a,(nn)
;;  S <- a      ld (nn),a
(define_insn "*movqi"
  [(set (match_operand:QI 0 "nonimmediate_operand" "=r,r,r,Q,Q,a,S")
	(match_operand:QI 1 "general_operand"      " r,n,Q,r,n,S,a"))]
  "z80_move_ok_p (operands[0], operands[1])"
  { return z80_output_movqi (operands); })

;; =========================================================================
;; 16-bit moves
;; =========================================================================

(define_expand "movhi"
  [(set (match_operand:HI 0 "nonimmediate_operand")
	(match_operand:HI 1 "general_operand"))]
  ""
{
  if (can_create_pseudo_p ()
      && MEM_P (operands[0])
      && !register_operand (operands[1], HImode)
      && !(CONST_INT_P (operands[1]) && z80_q_memory_p (operands[0])))
    operands[1] = force_reg (HImode, operands[1]);
})

;; "r" in HImode means one of the pairs BC, DE, HL.
(define_insn "*movhi"
  [(set (match_operand:HI 0 "nonimmediate_operand" "=r,r,r,Q,Q,r,S,r,k,y,k,r")
	(match_operand:HI 1 "general_operand"      " r,i,Q,r,n,S,r,x,r,r,i,k"))]
  "z80_move_ok_p (operands[0], operands[1])"
  { return z80_output_movhi (operands); })

;; =========================================================================
;; Pushes (used for passing arguments on the stack)
;; =========================================================================

(define_insn "pushhi1"
  [(set (mem:HI (pre_dec:HI (reg:HI SP_REG)))
	(match_operand:HI 0 "register_operand" "r"))]
  ""
  "push\t%0")

;; A byte argument occupies a 2-byte slot with the value in the low
;; (first) byte.  A push stores the high register at the higher address,
;; so a low register (C, E, L) is pushed directly; for the others the
;; stack pointer is offset by one byte around the push.
(define_insn "pushqi1"
  [(set (mem:HI (pre_dec:HI (reg:HI SP_REG)))
	(unspec:HI [(match_operand:QI 0 "register_operand" "l,r")]
		   UNSPEC_PUSHQI))]
  ""
  "@
   push\t%P0
   dec\tsp\;push\t%P0\;inc\tsp")

;; =========================================================================
;; 8-bit arithmetic and logic
;; =========================================================================

(define_insn "addqi3"
  [(set (match_operand:QI 0 "register_operand"          "=r,r,a")
	(plus:QI (match_operand:QI 1 "register_operand" "%0,0,0")
		 (match_operand:QI 2 "general_operand"  " I,J,rnQ")))]
  ""
  "@
   inc\t%0
   dec\t%0
   add\ta,%2")

(define_insn "subqi3"
  [(set (match_operand:QI 0 "register_operand"           "=a")
	(minus:QI (match_operand:QI 1 "register_operand" " 0")
		  (match_operand:QI 2 "general_operand"  " rnQ")))]
  ""
  "sub\t%2")

(define_insn "andqi3"
  [(set (match_operand:QI 0 "register_operand"         "=a")
	(and:QI (match_operand:QI 1 "register_operand" "%0")
		(match_operand:QI 2 "general_operand"  " rnQ")))]
  ""
  "and\t%2")

(define_insn "iorqi3"
  [(set (match_operand:QI 0 "register_operand"         "=a")
	(ior:QI (match_operand:QI 1 "register_operand" "%0")
		(match_operand:QI 2 "general_operand"  " rnQ")))]
  ""
  "or\t%2")

(define_insn "xorqi3"
  [(set (match_operand:QI 0 "register_operand"         "=a")
	(xor:QI (match_operand:QI 1 "register_operand" "%0")
		(match_operand:QI 2 "general_operand"  " rnQ")))]
  ""
  "xor\t%2")

(define_insn "negqi2"
  [(set (match_operand:QI 0 "register_operand"         "=a")
	(neg:QI (match_operand:QI 1 "register_operand" " 0")))]
  ""
  "neg")

(define_insn "one_cmplqi2"
  [(set (match_operand:QI 0 "register_operand"         "=a")
	(not:QI (match_operand:QI 1 "register_operand" " 0")))]
  ""
  "cpl")

;; =========================================================================
;; 16-bit arithmetic
;; =========================================================================

;; Alternatives:
;;   0  HL += BC/DE/HL           add hl,rr
;;   1  HL += SP                 add hl,sp
;;   2  rr += small constant     inc/dec rr
;;   3  rr += constant           via a temporary pair saved with push/pop
;;   4  rr += rr'                via HL (ex de,hl / push/pop)
;;   5  rr  = IX + constant      address of a stack slot
;;   6  SP += constant           stack adjustment (uses IY)
(define_insn "addhi3"
  [(set (match_operand:HI 0 "register_operand"          "=h,h,r,r,r,r,k")
	(plus:HI (match_operand:HI 1 "register_operand" "%0,0,0,0,0,x,0")
		 (match_operand:HI 2 "nonmemory_operand" "r,k,L,n,r,n,n")))]
  ""
  { return z80_output_addhi3 (operands); })

(define_insn "subhi3"
  [(set (match_operand:HI 0 "register_operand"           "=r")
	(minus:HI (match_operand:HI 1 "register_operand" " 0")
		  (match_operand:HI 2 "register_operand" " r")))]
  ""
  { return z80_output_subhi3 (operands); })

;; =========================================================================
;; Extensions
;; =========================================================================

(define_insn "zero_extendqihi2"
  [(set (match_operand:HI 0 "register_operand"                 "=r,r")
	(zero_extend:HI (match_operand:QI 1 "nonimmediate_operand" "r,Q")))]
  ""
{
  if (REG_P (operands[1]) && REGNO (operands[1]) == REGNO (operands[0]))
    return "ld\t%H0,0";
  return "ld\t%L0,%1\;ld\t%H0,0";
})

;; Copy the sign bit into the carry and use "sbc a,a" to get 0 or 0xff.
(define_insn "extendqihi2"
  [(set (match_operand:HI 0 "register_operand"                 "=r")
	(sign_extend:HI (match_operand:QI 1 "register_operand" "r")))
   (clobber (reg:QI A_REG))]
  ""
  "ld\ta,%1\;ld\t%L0,a\;rla\;sbc\ta,a\;ld\t%H0,a")

;; =========================================================================
;; Shifts
;; =========================================================================

(define_code_iterator any_shift [ashift lshiftrt ashiftrt])
(define_code_attr shift_name [(ashift "ashl") (lshiftrt "lshr")
			      (ashiftrt "ashr")])

;; Shift by a constant: an unrolled sequence of single-bit shifts.
;; Shift by a variable: a loop counting down a scratch register.
(define_insn "<shift_name>qi3"
  [(set (match_operand:QI 0 "register_operand"                "=r,r")
	(any_shift:QI (match_operand:QI 1 "register_operand"  " 0,0")
		      (match_operand:QI 2 "nonmemory_operand" " n,r")))
   (clobber (match_scratch:QI 3                               "=X,&r"))]
  ""
  { return z80_output_shift (<CODE>, operands); })

(define_insn "<shift_name>hi3"
  [(set (match_operand:HI 0 "register_operand"                "=r,r")
	(any_shift:HI (match_operand:HI 1 "register_operand"  " 0,0")
		      (match_operand:QI 2 "nonmemory_operand" " n,r")))
   (clobber (match_scratch:QI 3                               "=X,&r"))]
  ""
  { return z80_output_shift (<CODE>, operands); })

;; =========================================================================
;; Compare and branch
;; =========================================================================

;; The flags are not modelled, so the comparison and the conditional jump
;; are a single insn.  8-bit comparisons use "cp" with the first operand in
;; A; 16-bit comparisons use "sbc hl,rr", but accept any pair as first
;; operand (it is swapped/copied into HL inside the insn).  Requiring HL
;; itself makes LRA loop when HL holds a live hard-register value (e.g. a
;; return value reused as the next call's argument) across the jump.

;; 8-bit: the first operand in A, or any register when comparing with 0
;; ("inc r / dec r" sets the flags without needing A).
(define_insn "cbranchqi4"
  [(set (pc)
	(if_then_else
	  (match_operator 0 "ordered_comparison_operator"
	    [(match_operand:QI 1 "register_operand" "a,r")
	     (match_operand:QI 2 "general_operand"  "rnQ,P")])
	  (label_ref (match_operand 3 "" ""))
	  (pc)))]
  ""
  { return z80_output_cbranch (operands, false); })

(define_insn "*cbranchqi4_reversed"
  [(set (pc)
	(if_then_else
	  (match_operator 0 "ordered_comparison_operator"
	    [(match_operand:QI 1 "register_operand" "a,r")
	     (match_operand:QI 2 "general_operand"  "rnQ,P")])
	  (pc)
	  (label_ref (match_operand 3 "" ""))))]
  ""
  { return z80_output_cbranch (operands, true); })

;; 16-bit comparisons with zero use "ld a,h / or l" and therefore clobber A.
(define_expand "cbranchhi4"
  [(set (pc)
	(if_then_else
	  (match_operator 0 "ordered_comparison_operator"
	    [(match_operand:HI 1 "register_operand")
	     (match_operand:HI 2 "nonmemory_operand")])
	  (label_ref (match_operand 3 ""))
	  (pc)))]
  ""
{
  if (operands[2] == const0_rtx)
    {
      emit_jump_insn (gen_cbranchhi4_zero (operands[0], operands[1],
					   operands[3]));
      DONE;
    }
})

(define_insn "*cbranchhi4"
  [(set (pc)
	(if_then_else
	  (match_operator 0 "ordered_comparison_operator"
	    [(match_operand:HI 1 "register_operand"  "r,r")
	     (match_operand:HI 2 "nonmemory_operand" "r,i")])
	  (label_ref (match_operand 3 "" ""))
	  (pc)))]
  "operands[2] != const0_rtx"
  { return z80_output_cbranch (operands, false); })

(define_insn "*cbranchhi4_reversed"
  [(set (pc)
	(if_then_else
	  (match_operator 0 "ordered_comparison_operator"
	    [(match_operand:HI 1 "register_operand"  "r,r")
	     (match_operand:HI 2 "nonmemory_operand" "r,i")])
	  (pc)
	  (label_ref (match_operand 3 "" ""))))]
  "operands[2] != const0_rtx"
  { return z80_output_cbranch (operands, true); })

(define_insn "cbranchhi4_zero"
  [(set (pc)
	(if_then_else
	  (match_operator 0 "ordered_comparison_operator"
	    [(match_operand:HI 1 "register_operand" "r")
	     (const_int 0)])
	  (label_ref (match_operand 2 "" ""))
	  (pc)))
   (clobber (reg:QI A_REG))]
  ""
{
  operands[3] = operands[2];
  operands[2] = const0_rtx;
  return z80_output_cbranch (operands, false);
})

(define_insn "*cbranchhi4_zero_reversed"
  [(set (pc)
	(if_then_else
	  (match_operator 0 "ordered_comparison_operator"
	    [(match_operand:HI 1 "register_operand" "r")
	     (const_int 0)])
	  (pc)
	  (label_ref (match_operand 2 "" ""))))
   (clobber (reg:QI A_REG))]
  ""
{
  operands[3] = operands[2];
  operands[2] = const0_rtx;
  return z80_output_cbranch (operands, true);
})

(define_insn "jump"
  [(set (pc) (label_ref (match_operand 0 "" "")))]
  ""
  "jp\t%l0")

(define_insn "indirect_jump"
  [(set (pc) (match_operand:HI 0 "register_operand" "h"))]
  ""

;; Jump through a table of absolute addresses (switch statements).  GCC
;; loads the table entry; we only have to jump to it.
(define_insn "tablejump"
  [(set (pc) (match_operand:HI 0 "register_operand" "h"))
   (use (label_ref (match_operand 1 "" "")))]
  ""
  "jp\t(hl)")

;; =========================================================================
;; Calls
;; =========================================================================

;; Indirect calls load the target into BC (which is never used for
;; arguments) and use a local trampoline: "call" pushes the return address,
;; then "push bc / ret" enters the callee.

(define_expand "call"
  [(call (match_operand:QI 0 "memory_operand")
	 (match_operand 1 ""))]
  ""
{
  rtx addr = XEXP (operands[0], 0);
  if (!CONSTANT_P (addr))
    operands[0] = gen_rtx_MEM (QImode, force_reg (HImode, addr));
})

(define_insn "*call_direct"
  [(call (mem:QI (match_operand:HI 0 "immediate_operand" "i"))
	 (match_operand 1 "" ""))]
  ""
  "call\t%0")

(define_insn "*call_indirect"
  [(call (mem:QI (match_operand:HI 0 "register_operand" "b"))
	 (match_operand 1 "" ""))]
  ""
  "call\t.Lc%=\;jr\t.Ld%=\n.Lc%=:\n\tpush\tbc\n\tret\n.Ld%=:")

(define_expand "call_value"
  [(set (match_operand 0 "")
	(call (match_operand:QI 1 "memory_operand")
	      (match_operand 2 "")))]
  ""
{
  rtx addr = XEXP (operands[1], 0);
  if (!CONSTANT_P (addr))
    operands[1] = gen_rtx_MEM (QImode, force_reg (HImode, addr));
})

(define_insn "*call_value_direct"
  [(set (match_operand 0 "" "")
	(call (mem:QI (match_operand:HI 1 "immediate_operand" "i"))
	      (match_operand 2 "" "")))]
  ""
  "call\t%1")

(define_insn "*call_value_indirect"
  [(set (match_operand 0 "" "")
	(call (mem:QI (match_operand:HI 1 "register_operand" "b"))
	      (match_operand 2 "" "")))]
  ""
  "call\t.Lc%=\;jr\t.Ld%=\n.Lc%=:\n\tpush\tbc\n\tret\n.Ld%=:")

;; push ix / ld ix,0 / add ix,sp / allocate the frame.
(define_insn "z80_prologue"
  [(unspec_volatile [(match_operand:HI 0 "const_int_operand" "n")]
		    UNSPECV_PROLOGUE)
   (clobber (reg:HI IYL_REG))]
  ""
  { return z80_output_prologue (operands); })

(define_insn "z80_epilogue"
  [(unspec_volatile [(const_int 0)] UNSPECV_EPILOGUE)]
  ""
  "ld\tsp,ix\;pop\tix")

  ""
